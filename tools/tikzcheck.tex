% The TikZ side of tools/tikzcheck.lua: draws species \tikzcheckfrom..\tikzcheckto (role
% any), or the "seed role" lines of the file \tikzchecklist, with the pure-TikZ library
% and writes each one's choices, one line per species, to <jobname>.choices in the output
% directory. Runs under pdflatex and lualatex.
\documentclass{article}
\usepackage{tikz}
\usetikzlibrary{florilegium}
\begin{document}
\makeatletter
\newwrite\tikzcheckout
\immediate\openout\tikzcheckout=\jobname.choices
\florilegiumspecieslogtrue
\ifdefined\tikzchecklist
  \newread\tikzcheckin
  \openin\tikzcheckin=\tikzchecklist\relax
  \def\tikzcheckpair#1 #2\relax{\florilegiumspecies{#1}{#2}}
  {\endlinechar=-1
  \loop
    \read\tikzcheckin to \tikzcheckline
    \unless\ifx\tikzcheckline\empty
      \expandafter\tikzcheckpair\tikzcheckline\relax
      \immediate\write\tikzcheckout{\florilegiumspecieslog}%
    \fi
  \unless\ifeof\tikzcheckin\repeat}
\else
  \newcount\tikzcheckseed
  \tikzcheckseed=\tikzcheckfrom\relax
  \loop
    \florilegiumspecies{\the\tikzcheckseed}{any}%
    \immediate\write\tikzcheckout{\florilegiumspecieslog}%
    \advance\tikzcheckseed by 1
  \ifnum\tikzcheckseed>\tikzcheckto\relax\else\repeat
\fi
\immediate\closeout\tikzcheckout
\makeatother
Species written to \jobname.choices.
\end{document}
